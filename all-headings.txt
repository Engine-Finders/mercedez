# ============================================================
# 2) MODEL PAGE  (src/app/[...slug]/page.js — single segment)
# ============================================================
# H1 (dynamic: data.h1 from src/data/models/*.json)
#   Example: "A-Class Engines — The Complete UK Guide"
#   Pattern: "{Model} Engines — The Complete UK Guide"

# H2 — all dynamic (data.h2, except EditorialPullQuote uses data.title)
# Section order in page.js:
- Ownership Verdict        (data.h2)   e.g. "A-Class — The Ownership Verdict"
- At a Glance              (data.h2)   e.g. "A-Class at a Glance"
- Generations Grid         (data.h2)   e.g. "A-Class Generations — Find Your Vehicle"
- Engine Database          (data.h2)   e.g. "A-Class Engine Codes — The Complete Database"
- Common Problems          (data.h2)   e.g. "Common A-Class Engine Problems"
- Market Intelligence      (data.h2)   e.g. "A-Class Market Intelligence — What 750+ UK Owners Told Us in 2025"
- Editorial Pull Quote     (data.title) e.g. "The One Thing We'd Tell Every A-Class Buyer"
- Replacement Costs        (data.h2)   e.g. "A-Class Engine Replacement Costs — By Generation"
- Engine Evolution         (data.h2)   e.g. "How the A‑Class Engine Range Evolved — 1997 to Today"
- Who Should Buy           (data.h2)   e.g. "Who Should Buy an A‑Class?"
- Calculator CTA           (data.h2)   e.g. "Can I Repair My A‑Class Engine — or Should I Replace It?"
- Trust Block              (data.h2)   e.g. "Why A‑Class Owners Trust MercedesEngine.uk"
- FAQ                      (data.h2)   e.g. "A‑Class Frequently Asked Questions"
- Closing Action Cards     (data.h2)   e.g. "Ready to Make a Decision?"

# 3) GENERATION PAGE  (src/app/[...slug]/page.js — two segments)
# ============================================================
# H1 (dynamic: data.h1 from src/data/generations/*.json)
#   Example: "Mercedes A-Class W176 Engines — The Complete UK Guide"
#   Pattern: "{Generation} Engines — The Complete UK Guide"

# H2 — mix of static (hardcoded) and dynamic (data.h2)
# Section order in page.js:
- Engine Database          (data.h2)   e.g. "Mercedes A-Class W176 Engine Codes — The Complete Database"
- Overview                 (data.h2)   e.g. "Mercedes A-Class W176 — Overview"
- Best & Worst Engines     (STATIC)    "Best & Worst Engines"
- Ownership Economics      (data.h2)   e.g. "Mercedes A-Class W176 — Ownership Economics"
- Common Problems          (STATIC)    "Common Problems"
- Replacement Costs        (data.h2)   e.g. "W176 Engine Replacement Costs — By Engine and Condition"
- Core Variants            (STATIC)    "Core Variants"
- Market Intelligence      (STATIC)    "Market Intelligence"
- FAQ                      (STATIC)    "FAQ"
- Trust CTA                (data.h2)   e.g. "Why UK Mercedes A-Class W176 Owners Trust MercedesEngines.uk"


# ============================================================
# 4) VARIANT PAGE  (src/app/[...slug]/page.js — two segments)
# ============================================================
# H1 (dynamic: data.h1 from src/data/variants/*.json)
#   Example: "Mercedes‑Benz GLC‑Class GLC63 AMG Engine Replacement — Compare Prices & Get Quotes"

# H2 — mix of static and dynamic
# Section order in page.js:
- Era Map                  (data.title)  e.g. "The GLC63 AMG THROUGH TIME — Era Map"

- Engine Codes             (STATIC)      "Engine Codes"
- Market Intelligence      (STATIC)      "Market Intelligence"
- FAQ                      (STATIC)      "FAQ"
- Trust CTA                (data.h2)     e.g. "Why UK Mercedes‑Benz GLC63 AMG Owners Trust MercedesEngine.uk"
